// ================= CONFIGURACIÓN AVANZADA =================
const CONFIG = {
    MAX_USERS: 2000,              // Máximo de seguidores a recolectar
    MIN_WAIT_MS: 2500,           // Mínimo tiempo de espera (más humano)
    MAX_WAIT_MS: 7000,           // Máximo tiempo de espera
    SCROLL_INCREMENT: 400,       // Scroll suave en incrementos
    MAX_STAGNANT_ATTEMPTS: 3,    // Intentos antes de detener
    BATCH_SIZE: 50,              // Guardar en lotes para evitar pérdidas
    SAVE_EVERY_X_USERS: 100      // Guardar automático cada X usuarios
};

// ================= UTILIDADES =================
class InstagramScraper {
    constructor(config) {
        this.config = config;
        this.data = new Map(); // Usamos Map para mejor performance
        this.isRunning = false;
        this.startTime = null;
        this.batchCounter = 0;
        this.ui = this.createUI();
    }

    // Crear interfaz visual para monitoreo
    createUI() {
        const container = document.createElement('div');
        container.style.cssText = `
            position: fixed;
            top: 20px;
            right: 20px;
            background: rgba(0, 0, 0, 0.9);
            color: white;
            padding: 15px;
            border-radius: 10px;
            z-index: 999999;
            font-family: Arial, sans-serif;
            min-width: 250px;
            border: 2px solid #405de6;
        `;
        
        container.innerHTML = `
            <h3 style="margin:0 0 10px 0;color:#E1306C">🛡️ IG Scraper Seguro</h3>
            <div id="scraper-stats">
                <div>✅ Recolectados: <span id="count">0</span></div>
                <div>⏱️ Tiempo: <span id="time">00:00</span></div>
                <div>📈 Velocidad: <span id="speed">0/min</span></div>
                <div id="status" style="color:#00ff00">▶️ Iniciando...</div>

            <button id="save-btn" style="margin-top:5px;padding:5px 10px;background:#405de6;color:white;border:none;border-radius:5px;cursor:pointer">
                💾 Guardar Ahora
            </button>
        `;
        
        document.body.appendChild(container);
        return container;
    }

    updateUI(count) {
        const elapsed = Math.floor((Date.now() - this.startTime) / 1000);
        const minutes = Math.floor(elapsed / 60);
        const seconds = elapsed % 60;
        const speed = elapsed > 0 ? Math.floor((count / elapsed) * 60) : 0;
        
        this.ui.querySelector('#count').textContent = count;
        this.ui.querySelector('#time').textContent = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
        this.ui.querySelector('#speed').textContent = `${speed}/min`;
    }

    sleep(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }

    randomSleep() {
        const ms = Math.floor(Math.random() * 
            (this.config.MAX_WAIT_MS - this.config.MIN_WAIT_MS + 1)) + 
            this.config.MIN_WAIT_MS;
        return this.sleep(ms);
    }

    // Encontrar el contenedor scrollable de manera más robusta
    findScrollableContainer() {
        const selectors = [
            'div[role="dialog"] div[style*="overflow"]',
            'div[role="dialog"] > div > div',
            'div[role="dialog"] ._aano',
            'div[role="dialog"] div[class*="scroll"]'
        ];

        for (const selector of selectors) {
            const element = document.querySelector(selector);
            if (element && this.isScrollable(element)) {
                return element;
            }
        }

        // Fallback: buscar por comportamiento de scroll
        const dialog = document.querySelector('div[role="dialog"]');
        if (dialog) {
            const divs = dialog.querySelectorAll('div');
            for (const div of divs) {
                if (this.isScrollable(div)) {
                    return div;
                }
            }
        }

        throw new Error("No se encontró el contenedor scrollable");
    }

    isScrollable(element) {
        const style = window.getComputedStyle(element);
        const hasScroll = style.overflowY === 'auto' || style.overflowY === 'scroll';
        const hasHeight = element.scrollHeight > element.clientHeight;
        return hasScroll && hasHeight;
    }

    // Scroll suave con aceleración/desaceleración
    async smoothScroll(container) {
        const start = container.scrollTop;
        const end = container.scrollHeight;
        const distance = end - start;
        const duration = 800 + Math.random() * 400;
        const startTime = Date.now();

        return new Promise(resolve => {
            const animate = () => {
                const currentTime = Date.now();
                const elapsed = currentTime - startTime;
                const progress = Math.min(elapsed / duration, 1);
                
                // Función de easing (suavizado)
                const ease = 1 - Math.pow(1 - progress, 3);
                container.scrollTop = start + (distance * ease);

                if (progress < 1) {
                    requestAnimationFrame(animate);
                } else {
                    resolve();
                }
            };
            animate();
        });
    }

    // Extracción inteligente de datos
    extractUsers(container) {
        const users = new Set();
        
        // Selectores prioritarios para Instagram
        const selectors = [
            'a[href^="/"][tabindex="0"]',
            'div[role="dialog"] a[href^="/"]:not([href*="/p/"])',
            'a.x1i10hfl'
        ];

        for (const selector of selectors) {
            const elements = container.querySelectorAll(selector);
            if (elements.length > 0) {
                elements.forEach(el => {
                    const href = el.getAttribute('href');
                    if (href && href !== '/' && !href.includes('/p/')) {
                        const username = href.replace(/\//g, '').split('?')[0];
                        if (username && username.length > 1) {
                            // Extraer nombre real
                            const nameElement = el.querySelector('span, div');
                            const fullName = nameElement ? 
                                nameElement.textContent.trim().split('\n')[0] : 
                                "Sin Nombre";
                            
                            if (!this.data.has(username)) {
                                users.add(`${username},${fullName}`);
                            }
                        }
                    }
                });
                break; // Usar el primer selector que funcione
            }
        }

        return Array.from(users);
    }

    // Guardar progreso automáticamente
    autoSave() {
        this.batchCounter++;
        if (this.batchCounter % this.config.SAVE_EVERY_X_USERS === 0) {
            this.saveBackup();
        }
    }

    saveBackup() {
        const data = Array.from(this.data.values());
        const blob = new Blob([data.join('\n')], { type: 'text/csv' });
        const url = URL.createObjectURL(blob);
        
        // Solo guardamos en localStorage como backup
        try {
            localStorage.setItem('ig_scraper_backup', JSON.stringify({
                timestamp: Date.now(),
                count: data.length,
                data: data.slice(-100) // Últimos 100 como respaldo
            }));
            console.log(`💾 Backup guardado (${data.length} usuarios)`);
        } catch (e) {
            console.warn("No se pudo guardar backup en localStorage");
        }
        
        URL.revokeObjectURL(url);
    }

    // Descargar CSV final
    downloadCSV() {
        const data = Array.from(this.data.values());
        if (data.length === 0) {
            alert("No hay datos para descargar");
            return;
        }

        const headers = "Usuario,Nombre,Timestamp\n";
        const timestamp = new Date().toISOString();
        const content = headers + data.map(user => 
            `${user},${timestamp}`
        ).join("\n");

        const blob = new Blob([content], { type: 'text/csv;charset=utf-8;' });
        const link = document.createElement('a');
        const url = URL.createObjectURL(blob);
        
        link.setAttribute('href', url);
        link.setAttribute('download', `instagram_followers_${Date.now()}.csv`);
        link.style.visibility = 'hidden';
        
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        URL.revokeObjectURL(url);

        console.log(`✅ Descargado CSV con ${data.length} usuarios`);
    }

    // Ejecución principal
    async start() {
        if (this.isRunning) return;
        
        try {
            this.isRunning = true;
            this.startTime = Date.now();
            this.ui.querySelector('#status').textContent = "🔄 Ejecutando...";
            this.ui.querySelector('#status').style.color = "#00ff00";

            const container = this.findScrollableContainer();
            console.log("✅ Contenedor encontrado:", container);

            let stagnantAttempts = 0;
            let previousHeight = container.scrollHeight;
            let previousCount = 0;

            while (this.data.size < this.config.MAX_USERS && this.isRunning) {
                // Scroll suave
                await this.smoothScroll(container);
                
                // Espera humana variable
                await this.randomSleep();
                
                // Extraer usuarios
                const newUsers = this.extractUsers(container);
                newUsers.forEach(user => {
                    const [username] = user.split(',');
                    this.data.set(username, user);
                });

                // Actualizar UI
                this.updateUI(this.data.size);
                
                // Guardar progreso automático
                if (this.data.size > previousCount) {
                    this.autoSave();
                    previousCount = this.data.size;
                }

                // Verificar fin de lista
                if (container.scrollHeight === previousHeight) {
                    stagnantAttempts++;
                    console.log(`⚠️ Sin cambios (${stagnantAttempts}/${this.config.MAX_STAGNANT_ATTEMPTS})`);
                    
                    if (stagnantAttempts >= this.config.MAX_STAGNANT_ATTEMPTS) {
                        this.ui.querySelector('#status').textContent = "🏁 Lista completa";
                        break;
                    }
                } else {
                    stagnantAttempts = 0;
                    previousHeight = container.scrollHeight;
                }

                // Verificar límite
                if (this.data.size >= this.config.MAX_USERS) {
                    this.ui.querySelector('#status').textContent = "🎯 Límite alcanzado";
                    break;
                }
            }

            // Finalización
            if (this.isRunning) {
                this.ui.querySelector('#status').textContent = "✅ Proceso completado";
                this.ui.querySelector('#status').style.color = "#00ff00";
                this.downloadCSV();
            }

        } catch (error) {
            console.error("❌ Error:", error);
            this.ui.querySelector('#status').textContent = `❌ Error: ${error.message}`;
            this.ui.querySelector('#status').style.color = "#ff0000";
        } finally {
            this.isRunning = false;
        }
    }

    pause() {
        this.isRunning = false;
        this.ui.querySelector('#status').textContent = "⏸️ Pausado";
        this.ui.querySelector('#status').style.color = "#ffff00";
    }
}

        return;
    }

    // Verificar que la ventana de seguidores esté abierta
    if (!document.querySelector('div[role="dialog"]')) {
        alert("⚠️ Abre primero la ventana de seguidores/seguidos");
        return;
    }

    // Crear instancia del scraper
    const scraper = new InstagramScraper(CONFIG);

    // Configurar botones
    scraper.ui.querySelector('#pause-btn').onclick = () => {
        if (scraper.isRunning) {
            scraper.pause();
            scraper.ui.querySelector('#pause-btn').textContent = "▶️ Continuar";
        } else {
            scraper.ui.querySelector('#pause-btn').textContent = "⏸️ Pausar";
            scraper.start();
        }
    };

    scraper.ui.querySelector('#save-btn').onclick = () => {
        scraper.downloadCSV();
    };

    // Iniciar automáticamente después de 2 segundos
    await scraper.sleep(2000);
    scraper.start();

    return scraper;
}

// CORRECCIÓN: Monitor de Riesgos (Compatible con JS moderno)
const warningMonitor = {
    warnings: [],
    
    checkForWarnings() {
        // Buscamos texto específico en toda la página usando XPath (más seguro que selectores CSS)
        const warningTexts = [
            "Try Again Later",
            "Too Many Requests",
            "Action Blocked",
            "Intentar de nuevo más tarde", // Versión español
            "Acción bloqueada"             // Versión español
        ];

        let foundWarning = false;
        
        // Buscar en el cuerpo del documento
        const bodyText = document.body.innerText;
        
        warningTexts.forEach(text => {
            if (bodyText.includes(text)) {
                foundWarning = true;
                this.warnings.push({
                    time: Date.now(),
                    message: text
                });
            }
        });

        // También buscamos roles de alerta específicos
        const alertRoles = document.querySelectorAll('div[role="alert"]');
        if (alertRoles.length > 0) {
            foundWarning = true;
        }
        
        if (foundWarning && this.warnings.length >= 2) {
            console.warn("⚠️ MÚLTIPLES ADVERTENCIAS - DETENIENDO POR SEGURIDAD");
            // Asumimos que la variable 'scraper' es global o accesible
            if (typeof scraper !== 'undefined') scraper.pause();
            alert("Instagram ha detectado actividad inusual. Espera 24 horas.");
            return false;
        }
        return true;
    }
};

// Verificar cada minuto
